#!/usr/bin/env bash
# Misst kalten Bau + test --no-run je Variante. Aufruf: messen.sh <name> [--config ...]...
set -euo pipefail
cd /home/user/lifeline-hub
name="$1"; shift
out=/tmp/claude-0/-home-user-lifeline-hub/9a7a608b-cabf-5180-9f55-ec40caa56f29/scratchpad/mess/$name.txt
rm -rf target
{
echo "== $name: $*"
s=$(date +%s)
cargo build --workspace --exclude lifeline-desktop "$@" 2>&1 | tail -2
m=$(date +%s)
cargo test --workspace --exclude lifeline-desktop --no-run "$@" 2>&1 | tail -2
e=$(date +%s)
echo "build_s=$((m-s)) test_no_run_s=$((e-m)) total_s=$((e-s))"
du -sm target target/debug/* | sort -k2
echo "testbinaries:"
n=0; sz=0
while read -r f; do n=$((n+1)); sz=$((sz+$(stat -c %s "$f"))); done < <(find target/debug/deps -maxdepth 1 -type f -executable ! -name '*.so' ! -name '*.d')
echo "anzahl=$n mb=$((sz/1024/1024))"
echo "lifeline-hub binary: $(stat -c %s target/debug/lifeline-hub | awk '{print int($1/1024/1024)}') MB"
} > "$out" 2>&1
cat "$out"
